cmake_minimum_required(VERSION 3.24)
project(hft_sdk_consumer LANGUAGES CXX)
option(HFT_CONSUMER_HEADER_ONLY "Use only the header-only target" OFF)
find_package(HftSdk 1.0 CONFIG REQUIRED)
add_executable(consumer main.cpp)
if(HFT_CONSUMER_HEADER_ONLY)
    target_compile_definitions(consumer PRIVATE HFT_CONSUMER_HEADER_ONLY=1)
    target_link_libraries(consumer PRIVATE HftSdk::core)
else()
    target_link_libraries(consumer PRIVATE HftSdk::sdk)
endif()
